<script>
  // ── CONFIGURATION ────────────────────────────────────────────────────────
  // Replace this URL with your Google Apps Script Web App URL after deployment
  const SCRIPT_URL = 'Yhttps://script.google.com/macros/s/AKfycby6U8TAHExf08FMhRryyzB83Y18M5BmOQ4uBYK0tl1f247oeGATi6jORmN5Vx9ERAzE/exec';
  // ────────────────────────────────────────────────────────────────────────

  // Auto date display
  const today = new Date();
  const dateStr = today.toLocaleDateString('en-GB', {
    weekday: 'long', day: 'numeric', month: 'long', year: 'numeric'
  });
  document.getElementById('auto-date').textContent = dateStr;

  // Remember rep name across sessions (localStorage)
  const repSelect = document.getElementById('rep-select');
  const savedRep = localStorage.getItem('solutech_rep');
  if (savedRep) repSelect.value = savedRep;

  function saveRep() {
    localStorage.setItem('solutech_rep', repSelect.value);
  }

  // Form submission
  async function submitForm(e) {
    e.preventDefault();

    const rep = repSelect.value;
    if (!rep) {
      showError('Please select your name at the top before submitting.');
      return;
    }

    const form    = document.getElementById('daily-form');
    const btn     = document.getElementById('submit-btn');
    const data    = new FormData(form);

    // Build payload
    const payload = {
      rep_name:            rep,
      date:                today.toLocaleDateString('en-GB'),   // DD/MM/YYYY
      timestamp:           today.toISOString(),
      region:              data.get('region'),
      route:               data.get('route'),
      sales_ex_vat:        data.get('sales'),
      customers_visited:   data.get('customers_visited') || 0,
      customers_purchased: data.get('customers_purchased') || 0,
      notes:               data.get('notes') || ''
    };

    // UI: loading state
    btn.disabled = true;
    btn.innerHTML = '<span class="spinner"></span>Submitting...';
    hideMessages();

    try {
      await fetch(SCRIPT_URL, {
        method: 'POST',
        mode: 'no-cors',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      // no-cors returns opaque response — assume success if no error thrown
      showSuccess();
      form.reset();
      repSelect.value = localStorage.getItem('solutech_rep') || '';
      document.getElementById('confirm-date').textContent =
        today.toLocaleDateString('en-GB', { day: 'numeric', month: 'long', year: 'numeric' });

    } catch (err) {
      showError('Network error. Check your connection and try again.');
    } finally {
      btn.disabled = false;
      btn.innerHTML = 'Submit Daily Report';
    }
  }

  function showSuccess() {
    document.getElementById('msg-success').style.display = 'block';
    document.getElementById('msg-error').style.display   = 'none';
  }
  function showError(msg) {
    document.getElementById('error-text').textContent    = msg;
    document.getElementById('msg-error').style.display  = 'block';
    document.getElementById('msg-success').style.display = 'none';
  }
  function hideMessages() {
    document.getElementById('msg-success').style.display = 'none';
    document.getElementById('msg-error').style.display   = 'none';
  }
</script>
